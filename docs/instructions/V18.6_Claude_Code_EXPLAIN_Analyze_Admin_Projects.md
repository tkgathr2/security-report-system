cd C:\Users\takag\00_dev\security-report-system\backend
node -e "require('dotenv').config(); const {Client}=require('pg'); (async()=>{ const c=new Client({connectionString:process.env.DATABASE_URL}); await c.connect(); const q=\`select tablename,indexname,indexdef from pg_indexes where schemaname='public' and tablename in ('projects','project_casts') order by tablename,indexname;\`; const r=await c.query(q); require('fs').writeFileSync('../docs/debug/indexes_projects_project_casts.json', JSON.stringify(r.rows,null,2)); console.log('saved: docs/debug/indexes_projects_project_casts.json'); await c.end(); })().catch(e=>{console.error(String(e)); process.exit(1);});"

## V0.3 EXPLAIN ANALYZE（ANALYZE 前）を保存

対象期間はこれまでと同じ：
- start: `2026-02-01`
- end: `2026-02-28`

以下を実行（EXPLAINは `FORMAT JSON` で保存。行データは出ない）：

```powershell
cd C:\Users\takag\00_dev\security-report-system\backend
node -e "require('dotenv').config(); const fs=require('fs'); const {Client}=require('pg'); (async()=>{ const c=new Client({connectionString:process.env.DATABASE_URL}); await c.connect(); const start='2026-02-01', end='2026-02-28'; const projectsSql=\`EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) SELECT p.id, p.project_key, p.client_name_raw, p.work_date, p.work_name, p.location, p.status, p.unique_url, p.url_expires_at, p.created_at, c.name as client_name FROM projects p LEFT JOIN clients c ON p.client_id = c.id WHERE p.work_date >= $1 AND p.work_date <= $2 ORDER BY p.work_date ASC, p.created_at DESC\`; const pr=await c.query(projectsSql,[start,end]); fs.writeFileSync('../docs/debug/explain_projects_before_analyze.json', JSON.stringify(pr.rows[0]['QUERY PLAN'],null,2)); const ids=await c.query('SELECT id FROM projects WHERE work_date >= $1 AND work_date <= $2 ORDER BY work_date ASC, created_at DESC LIMIT 200',[start,end]); const idArr=ids.rows.map(r=>r.id); const castsSql=\`EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) SELECT project_id, staff_no, cast_name FROM project_casts WHERE project_id = ANY($1) ORDER BY row_index\`; const cr=await c.query(castsSql,[idArr]); fs.writeFileSync('../docs/debug/explain_project_casts_before_analyze.json', JSON.stringify(cr.rows[0]['QUERY PLAN'],null,2)); console.log('saved: docs/debug/explain_projects_before_analyze.json'); console.log('saved: docs/debug/explain_project_casts_before_analyze.json'); await c.end(); })().catch(e=>{console.error(String(e)); process.exit(1);});"
```

完了条件：
- `docs/debug/explain_projects_before_analyze.json`
- `docs/debug/explain_project_casts_before_analyze.json`
が生成される
